# Cron Job Configuration for Data Ingestion Scripts
# Format: minute hour day month day_of_week command
# 
# Installation: Run setup_cron_jobs.sh to install these cron jobs
# Logs: /var/log/data-ingestion/

# Set environment variables
SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin
PYTHONPATH=/path/to/data-team/data-eng
GOOGLE_APPLICATION_CREDENTIALS=/path/to/credentials.json

# ============================================================================
# DAILY INGESTION JOBS
# ============================================================================

# FEWS NET - IPC Phase Data (Daily at 6:00 AM)
0 6 * * * /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh fews_net_ingestion.py --all-countries --data-type ipc_phase >> /var/log/data-ingestion/fews_net_ipc.log 2>&1

# FEWS NET - Market Prices (Daily at 6:30 AM)
30 6 * * * /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh fews_net_ingestion.py --all-countries --data-type market_prices >> /var/log/data-ingestion/fews_net_market.log 2>&1

# FEWS NET - Food Security (Daily at 7:00 AM)
0 7 * * * /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh fews_net_ingestion.py --all-countries --data-type food_security >> /var/log/data-ingestion/fews_net_food_security.log 2>&1

# WFP Food Prices (Daily at 7:30 AM)
30 7 * * * /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh wfp_food_prices_ingestion.py --all-countries >> /var/log/data-ingestion/wfp_prices.log 2>&1

# ============================================================================
# WEEKLY INGESTION JOBS (Sunday)
# ============================================================================

# ISRIC Soil Data - East Africa (Sunday at 2:00 AM)
0 2 * * 0 /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh isric_soil_ingestion.py --region east_africa --grid-resolution 0.5 >> /var/log/data-ingestion/isric_soil_east.log 2>&1

# ISRIC Soil Data - West Africa (Sunday at 3:00 AM)
0 3 * * 0 /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh isric_soil_ingestion.py --region west_africa --grid-resolution 0.5 >> /var/log/data-ingestion/isric_soil_west.log 2>&1

# ISRIC Soil Data - Southern Africa (Sunday at 4:00 AM)
0 4 * * 0 /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh isric_soil_ingestion.py --region southern_africa --grid-resolution 0.5 >> /var/log/data-ingestion/isric_soil_south.log 2>&1

# GBIF Biodiversity - East Africa (Sunday at 5:00 AM)
0 5 * * 0 /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh gbif_biodiversity_ingestion.py --region east_africa --limit 5000 >> /var/log/data-ingestion/gbif_east.log 2>&1

# GBIF Biodiversity - West Africa (Sunday at 6:00 AM)
0 6 * * 0 /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh gbif_biodiversity_ingestion.py --region west_africa --limit 5000 >> /var/log/data-ingestion/gbif_west.log 2>&1

# ============================================================================
# MONTHLY INGESTION JOBS (1st of month)
# ============================================================================

# FAO Crop Production - QC Domain (1st at 1:00 AM)
0 1 1 * * /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh fao_crop_ingestion.py --all-countries --domain QC >> /var/log/data-ingestion/fao_qc.log 2>&1

# FAO Crop Production - RL Domain (1st at 2:00 AM)
0 2 1 * * /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh fao_crop_ingestion.py --all-countries --domain RL >> /var/log/data-ingestion/fao_rl.log 2>&1

# FAO Crop Production - RFN Domain (1st at 3:00 AM)
0 3 1 * * /path/to/data-team/data-eng/scripts/run_ingestion_wrapper.sh fao_crop_ingestion.py --all-countries --domain RFN >> /var/log/data-ingestion/fao_rfn.log 2>&1

# ============================================================================
# MAINTENANCE JOBS
# ============================================================================

# Log rotation (Daily at midnight)
0 0 * * * find /var/log/data-ingestion/ -name "*.log" -mtime +30 -delete

# Health check (Every 6 hours)
0 */6 * * * /path/to/data-team/data-eng/scripts/check_ingestion_logs.py >> /var/log/data-ingestion/health_check.log 2>&1
